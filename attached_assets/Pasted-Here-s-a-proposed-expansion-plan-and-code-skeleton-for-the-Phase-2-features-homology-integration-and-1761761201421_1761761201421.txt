# homology/diamond_runner.py
import subprocess
from pathlib import Path

class DiamondRunner:
    def __init__(self, db_path, query_path, output_path, threads=8):
        self.db_path = Path(db_path)
        self.query_path = Path(query_path)
        self.output_path = Path(output_path)
        self.threads = threads

    def run(self):
        cmd = [
            "diamond", "blastp",
            "--db", str(self.db_path),
            "--query", str(self.query_path),
            "--out", str(self.output_path),
            "--outfmt", "6 qseqid sseqid bitscore evalue",
            "--max-target-seqs", "10",
            "--threads", str(self.threads)
        ]
        subprocess.run(cmd, check=True)
```

# homology/feature_extractor.py
import pandas as pd
import numpy as np

class HomologyFeatureExtractor:
    def __init__(self, diamond_output_path, score_method="bitscore"):
        self.diamond_output_path = diamond_output_path
        self.score_method = score_method

    def load_scores(self):
        df = pd.read_csv(self.diamond_output_path, sep='\t', names=["qseqid", "sseqid", "bitscore", "evalue"])
        agg = df.groupby("qseqid")[self.score_method].mean()
        return agg
```

import numpy as np

def merge_features(embeddings, homology_scores):
    merged = np.concatenate([embeddings, homology_scores.reshape(-1, 1)], axis=1)
    return merged

### Zero-Shot GO Encoder Skeleton

```python
# zero_shot/term_encoder.py
from transformers import AutoTokenizer, AutoModel
import torch

class GOTermEncoder:
    def __init__(self, model_name="sentence-transformers/all-MiniLM-L6-v2"):
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModel.from_pretrained(model_name)

    def encode_terms(self, terms):
        inputs = self.tokenizer(terms, padding=True, truncation=True, return_tensors="pt")
        with torch.no_grad():
            outputs = self.model(**inputs)
        return outputs.last_hidden_state.mean(dim=1)
```

```python
# zero_shot/similarity_scorer.py
import torch
import torch.nn.functional as F

def compute_similarity(protein_embeddings, term_embeddings):
    protein_norm = F.normalize(protein_embeddings, p=2, dim=1)
    term_norm = F.normalize(term_embeddings, p=2, dim=1)
    return torch.matmul(protein_norm, term_norm.T)
```

```python
# zero_shot/zero_shot_model.py
import torch

class ZeroShotModel(torch.nn.Module):
    def __init__(self, embedding_dim, term_embedding_dim):
        super().__init__()
        self.protein_proj = torch.nn.Linear(embedding_dim, term_embedding_dim)

    def forward(self, protein_embeddings, term_embeddings):
        projected = self.protein_proj(protein_embeddings)
        scores = torch.matmul(projected, term_embeddings.T)
        return scores
```
