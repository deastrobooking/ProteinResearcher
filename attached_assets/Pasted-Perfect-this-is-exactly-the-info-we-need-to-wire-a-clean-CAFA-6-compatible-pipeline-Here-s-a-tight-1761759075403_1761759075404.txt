```python
import obonet, networkx as nx, pandas as pd, numpy as np, re
from pathlib import Path

DATA = Path("data/")  # adjust as needed

ROOTS = {'BPO': 'GO:0008150', 'CCO': 'GO:0005575', 'MFO': 'GO:0003674'}

# --- Load GO and build a single DAG (child -> parent edges) ---
def load_go_graph(obo_path: Path):
    G = obonet.read_obo(str(obo_path))          # returns a MultiDiGraph of GO
    # Normalize to DiGraph with child->parent across 'is_a' and 'part_of'
    DG = nx.DiGraph()
    for n, d in G.nodes(data=True):
        if not n.startswith("GO:"): 
            continue
        DG.add_node(n, name=d.get('name'), namespace=d.get('namespace'))  # namespace: biological_process / cellular_component / molecular_function
        parents = []
        for rel in ('is_a', 'part_of'):
            for edge in d.get(rel, []):
                p = edge.split(' ! ')[0] if ' ! ' in edge else edge
                if p.startswith("GO:"): parents.append(p)
        for p in parents:
            DG.add_edge(n, p)  # child -> parent
    return DG

GO = load_go_graph(DATA / "go-basic.obo")

# --- Map GO IDs to subontologies (BPO/CCO/MFO) ---
def go_to_subonto(go_id: str) -> str | None:
    ns = GO.nodes[go_id].get('namespace', '')
    if ns == 'biological_process': return 'BPO'
    if ns == 'cellular_component': return 'CCO'
    if ns == 'molecular_function': return 'MFO'
    return None

# --- Build per-ontology vocabularies (exclude roots if you want; we’ll handle closure at score time) ---
def build_term_lists():
    by_onto = {'BPO': [], 'CCO': [], 'MFO': []}
    for n in GO.nodes:
        o = go_to_subonto(n)
        if o: by_onto[o].append(n)
    # Fix order for reproducibility
    for k in by_onto: by_onto[k] = sorted(set(by_onto[k]))
    return by_onto

TERMS = build_term_lists()  # dict[onto] -> [go_ids]

# --- Ancestor tables (indices) for fast closure ---
def ancestor_index(onto_terms):
    idx = {t:i for i,t in enumerate(onto_terms)}
    anc = {}
    for j,t in enumerate(onto_terms):
        # Get all ancestors (parents, grandparents...) within same ontology
        a = {a for a in nx.ancestors(GO, t) if a in idx}
        anc[j] = {idx[a] for a in a}
    return idx, anc

IDX = {}
ANC = {}
for onto in ('MFO','BPO','CCO'):
    IDX[onto], ANC[onto] = ancestor_index(TERMS[onto])
```

# 6) ic-weighted maxF1 (matches CAFA scoring style)

```python
def ancestor_close_scores(S: np.ndarray, anc: dict) -> np.ndarray:
    # S: [N, C] probabilities; propagate child→parent via max
    P = S.copy()
    # One simple topological order approach: repeatedly raise parents to max(child,parent)
    # Precompute parent lists for speed:
    parents_of = {j:list(anc[j]) for j in anc}
    changed = True
    # loop count is small in practice due to shallow paths
    while changed:
        changed = False
        for j, parents in parents_of.items():
            if not parents: continue
            parent_vals = P[:, parents]
            new_vals = np.maximum(parent_vals, P[:, [j]])
            if (new_vals > parent_vals).any():
                P[:, parents] = np.maximum(parent_vals, new_vals)
                changed = True
    return P

def maxF1_ic(S_raw, Y_true, ia_vec, anc, thresholds=np.linspace(0.01, 0.99, 99)):
    S = ancestor_close_scores(S_raw, anc)
    best = (0.0, 0.5)
    ia = ia_vec
    # Avoid divide-by-zero
    eps = 1e-12
    for t in thresholds:
        P = (S >= t).astype(np.uint8)
        tp = ((P & (Y_true==1)) * ia).sum()
        fp = ((P & (Y_true==0)) * ia).sum()
        fn = (((1-P) & (Y_true==1)) * ia).sum()
        wP = tp / (tp + fp + eps)
        wR = tp / (tp + fn + eps)
        F = (2*wP*wR) / (wP + wR + eps)
        if F > best[0]: best = (F, t)
    return best  # (maxF1, threshold)
```

Use this on a validation fold **per subontology** to pick thresholds, then report the mean of the three maxF1s.

```python
def read_fasta_ids(fasta_path: Path):
    ids, seqs = [], {}
    for rec in SeqIO.parse(str(fasta_path), "fasta"):
```

# 8) Submission builder (format-safe)

* Combine MF/BP/CC predictions in one file.
* **Do not** output zeros; cap **≤1500 terms** total per protein; **≤3 significant figures**; propagate to parents is done by organizers if you haven’t already, but it’s best to output children plus let parent propagation happen (we still do closure for scoring/thresholding).

```python
def to_sig3(x: float) -> str:
    # scores in (0,1], not 0; clamp tiny positives to 0.001
    x = float(np.clip(x, 1e-6, 1.0))
    return f"{x:.3g}"  # up to 3 significant figures

def build_submission(per_onto_scores: dict[str, dict[str, list[tuple[str,float]]]],
                     out_path: Path,
                     max_terms_per_protein: int = 1500):
    """
    per_onto_scores[onto][pid] = [(go_id, prob), ...] already filtered by thresholds & sorted desc
    """
    with out_path.open("w", encoding="utf-8") as f:
        for pid in test_ids:
            rows = []
            for onto in ('MFO','BPO','CCO'):
                for go, p in per_onto_scores.get(onto, {}).get(pid, []):
                    rows.append((go, p))
            # sort global desc and clip
            rows.sort(key=lambda x: x[1], reverse=True)
            rows = rows[:max_terms_per_protein]
            for go, p in rows:
                f.write(f"{pid}\t{go}\t{to_sig3(p)}\n")
```

You’ll fill `per_onto_scores` from your model outputs (pLM head, homology head, zero-shot head) **after** calibration & thresholding.
